def band_for(score: float, medium: float, high: float) -> str:
    if not (0.0 <= score <= 1.0):
        raise ValueError(f"score muss zwischen 0.0 und 1.0 liegen, erhalten: {score}")
    if not (0.0 <= medium <= 1.0):
        raise ValueError(f"medium muss zwischen 0.0 und 1.0 liegen, erhalten: {medium}")
    if not (0.0 <= high <= 1.0):
        raise ValueError(f"high muss zwischen 0.0 und 1.0 liegen, erhalten: {high}")
    if medium > high:
        raise ValueError(f"medium darf high nicht uebersteigen, erhalten: medium={medium}, high={high}")

    if score >= high:
        return 'hoch'
    if score >= medium:
        return 'mittel'
    return 'niedrig'
